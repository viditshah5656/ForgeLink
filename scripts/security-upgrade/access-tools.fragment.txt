      content: [{ type: "text", text: JSON.stringify(structuredContent, null, 2) }],

  server.registerTool("use_path_access_once", {
    description: "Execute exactly one owner-approved filesystem tool with a short-lived capability delivered to the current chat. The capability is bound to one exact directory scope, access level, and intended filesystem tool, and it is consumed before execution. It cannot enable terminal, project-process, Windows-control, approval, restart, or safety tools.",
    inputSchema: {
      capabilityToken: z.string().min(32).max(200),
      toolName: z.string().min(1).max(100),
      arguments: z.record(z.any()).optional().default({}),
    },
    annotations: { readOnlyHint: false, destructiveHint: true, openWorldHint: false },
    ...common,
  }, async ({ capabilityToken, toolName, arguments: toolArguments }) => {
    if (!ONE_TIME_ACCESS_ALLOWED_TOOLS.has(toolName)) {
      throw new Error("This tool is not eligible for one-time filesystem access.");
    }

    const targetHandler = registeredToolHandlers.get(toolName);
    if (!targetHandler || toolName === "use_path_access_once") {
      throw new Error("The bound filesystem tool is unavailable.");
    }

    const capability = claimAccessCapability(capabilityToken, toolName);
    await writeAudit("access.capability_consumed", {
      path: capability.root,
      access: capability.access,
      intendedTool: capability.intendedTool,
      requestId: capability.requestId,
    });

    return await accessCapabilityContext.run(
      capability,
      async () => await targetHandler(toolArguments || {}),
    );
  });
